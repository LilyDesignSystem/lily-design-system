قابلِ رسائی فارم ٹیوٹوریل

فریم ورک کوئی بھی ہو، ہر Lily فارم یہی طریقہ اپناتا ہے: ہر کنٹرول کو لیبل دیں، جمع کرانے پر جانچیں، ہر غلطی اس کے خانے کے ساتھ رکھیں، اور جہاں فوکس پہنچے وہاں غلطیوں کا خلاصہ دیں۔

آپ کیا بنائیں گے

دو خانوں والا ایک فارم، جو غلطیوں کے ساتھ جمع ہونے پر بتاتا ہے کہ کیا غلط ہے، ہر خانے کے ساتھ اور اوپر ایک خلاصے میں، اور فوکس وہیں لے جاتا ہے، تاکہ کی بورڈ یا اسکرین ریڈر استعمال کرنے والا فوراً غلطیاں ڈھونڈ لے۔ اسے Lily کلاس ہکس کے ساتھ سادہ HTML میں دکھایا گیا ہے؛ ہر فریم ورک کیٹلاگ اپنے Field، Hint، ErrorMessage اور ErrorSummary اجزاء سے یہی مارک اپ بناتا ہے۔ تقریباً 20 منٹ؛ پہلے HTML ٹیوٹوریل کر لینا مددگار ہے۔

مرحلہ 1 — ہر کنٹرول کو لیبل دیں، اور بتائیں کہ آپ کیا چاہتے ہیں

<form class="form" novalidate>
  <div class="field">
    <label class="label" for="name">Full name</label>
    <input class="text-input" type="text" id="name" name="name" autocomplete="name" required />
  </div>
  <div class="field">
    <label class="label" for="email">Email</label>
    <p class="hint" id="email-hint">We only use this to send your appointment.</p>
    <input class="email-input" type="email" id="email" name="email"
      autocomplete="email" aria-describedby="email-hint" required />
  </div>
  <button class="button" type="submit">Continue</button>
</form>
  • ہر کنٹرول پر ایک نظر آنے والا <label for> ، اس کی جگہ کبھی پلیس ہولڈر نہیں: پلیس ہولڈر ٹائپ کرتے ہی غائب ہو جاتا ہے۔
  • اشارہ خانے کے نام سے aria-describedbyکے ذریعے جڑتا ہے، اس لیے اسکرین ریڈر اسے خانے کے ساتھ پڑھتا ہے۔
  • autocomplete براؤزر کو وہ بھرنے دیتا ہے جو وہ جانتا ہے (WCAG 1.3.5)۔
  • novalidate براؤزر کے اپنے بلبلے بند کر دیتا ہے، کیونکہ غلطیاں آپ خود، اپنے الفاظ اور اپنی زبان میں دکھائیں گے۔ required اور type برقرار رہتے ہیں: آپ کا اسکرپٹ انہیں پڑھتا ہے۔

مرحلہ 2 — جمع کرانے پر جانچیں، ٹائپ کرتے وقت نہیں

ایک حرف کے بعد ہی کسی کو بتانا کہ اس کی ای میل غلط ہے، محض شور ہے۔ جمع کرانے پر جانچیں۔ تمام 63 تھیمز کسی خانے کو تبھی غلط دکھاتی ہیں جب اسے بدل کر چھوڑا گیا ہو (:user-invalid)، یا جب آپ aria-invalid="true"سیٹ کریں، اس لیے صفحہ کھلنے پر کچھ بھی سرخ نہیں ہوتا۔

فارم درست ہونے تک جمع کرانے کا بٹن کبھی غیر فعال نہ کریں: غیر فعال بٹن وجہ نہیں بتا سکتا۔

مرحلہ 3 — ہر غلطی اس کے خانے کے ساتھ رکھیں

<div class="field">
  <label class="label" for="email">Email</label>
  <p class="hint" id="email-hint">We only use this to send your appointment.</p>
  <p class="error-message" id="email-error">Enter an email address in the right format, like name@example.com</p>
  <input class="email-input" type="email" id="email" name="email" autocomplete="email"
    aria-invalid="true" aria-describedby="email-hint email-error" required />
</div>
  • aria-invalid="true" بتاتا ہے کہ خانہ غلط ہے؛ پیغام بتاتا ہے کیوں اور کیسے درست کریں۔
  • پیغام aria-describedbyمیں ہے، اشارے کے بعد، اس لیے خانے پر فوکس آنے پر پڑھا جاتا ہے۔
  • بتائیں کہ کیا کرنا ہے، صرف یہ نہیں کہ کیا غلط ہے: "درست شکل میں ای میل پتہ درج کریں"، نہ کہ "غلط"۔
  • کبھی صرف رنگ نہیں: متن ہی غلطی ہے؛ سرخ کنارہ بس اسے دہراتا ہے (WCAG 1.4.1)۔

مرحلہ 4 — جہاں فوکس پہنچے وہاں غلطیوں کا خلاصہ دیں

<div class="error-summary" role="alert" tabindex="-1" aria-labelledby="error-summary-title">
  <h2 id="error-summary-title">There is a problem</h2>
  <ul>
    <li><a href="#name">Enter your full name</a></li>
    <li><a href="#email">Enter an email address in the right format, like name@example.com</a></li>
  </ul>
</div>

اسے فارم کے اوپر رکھیں اور جمع کرانے پر فوکس اس پر لے جائیں۔ ہر لنک سیدھا اپنے خانے پر جاتا ہے۔ tabindex="-1" خلاصے کو نیا ٹیب پڑاؤ شامل کیے بغیر فوکس لینے دیتا ہے۔

مرحلہ 5 — سب کچھ جوڑنے والا اسکرپٹ

const messages = {
  name: { valueMissing: "Enter your full name" },
  email: {
    valueMissing: "Enter your email address",
    typeMismatch: "Enter an email address in the right format, like name@example.com",
  },
};

const form = document.querySelector("form");
form.addEventListener("submit", (event) => {
  form.querySelector(".error-summary")?.remove();
  const errors = [];
  for (const field of form.querySelectorAll("input")) {
    const old = document.getElementById(field.id + "-error");
    old?.remove();
    const describedBy = (field.getAttribute("aria-describedby") ?? "")
      .split(" ").filter((id) => id && id !== field.id + "-error");
    const problem = Object.keys(messages[field.name] ?? {}).find((key) => field.validity[key]);
    if (!problem) {
      field.removeAttribute("aria-invalid");
      field.setAttribute("aria-describedby", describedBy.join(" "));
      continue;
    }
    const text = messages[field.name][problem];
    const message = document.createElement("p");
    message.className = "error-message";
    message.id = field.id + "-error";
    message.textContent = text;
    field.before(message);
    field.setAttribute("aria-invalid", "true");
    field.setAttribute("aria-describedby", [...describedBy, message.id].join(" "));
    errors.push({ id: field.id, text });
  }
  if (errors.length === 0) return; // valid: let it submit
  event.preventDefault();
  const summary = document.createElement("div");
  summary.className = "error-summary";
  summary.setAttribute("role", "alert");
  summary.tabIndex = -1;
  summary.setAttribute("aria-labelledby", "error-summary-title");
  summary.innerHTML = '<h2 id="error-summary-title">There is a problem</h2><ul></ul>';
  for (const { id, text } of errors) {
    const link = document.createElement("a");
    link.href = "#" + id;
    link.textContent = text;
    const item = document.createElement("li");
    item.append(link);
    summary.querySelector("ul").append(item);
  }
  form.prepend(summary);
  summary.focus();
});

پیغامات آپ کے ہیں، آپ کی زبان میں: براؤزر کا validity بتاتا ہے کہ کون سا اصول ناکام ہوا، اور آپ کی جدول بتاتی ہے کہ فرد سے کیا کہنا ہے۔ Lily ترجمے کے لیے کوئی انگریزی متن نہیں دیتا۔

اپنا کام جانچیں: خالی فارم کی بورڈ سے جمع کرائیں۔ فوکس خلاصے پر جاتا ہے اور اسکرین ریڈر "There is a problem" اور دونوں غلطیاں پڑھتا ہے۔ کسی لنک پر جائیں: فوکس خانے میں پہنچتا ہے، جو اپنے اشارے اور غلطی کے ساتھ پڑھا جاتا ہے۔ دونوں درست کر کے جمع کرائیں: فارم چلا جاتا ہے، کوئی غلطی باقی نہیں رہتی۔

کن باتوں سے بچیں

  • کسی کے لکھے کو بتائے بغیر محدود یا درست کرنا؛ اس کے بجائے انہیں بتائیں۔
  • غلطی پر فارم خالی کر دینا: جو کچھ درج کیا گیا سب رکھیں۔
  • ایسی غلطی جو صرف ہوور پر یا صرف ٹول ٹپ میں دکھائی دے۔
  • بغیر خبردار کیے فارم کا وقت ختم ہونا؛ خبردار کرنے اور وقت بڑھانے کے لیے timeout-dialog استعمال کریں۔

اگلے اقدامات

  • تربیت — آٹھ ماڈیول، ہر ایک میں ایک مشق اور خود جانچ۔
  • ErrorSummary، Field اور ErrorMessage کیٹلاگ میں۔
  • دنیا بھر میں کام کرنے والا فارم — نام، پتے، تاریخیں اور لکھائی کی سمت۔
  • تھیمنگ — ہر تھیم جو غلط حالت دکھاتی ہے۔
